cmake_minimum_required(VERSION 3.16)
project(igris_sim_ctl)
find_package(ament_cmake REQUIRED)
find_package(igris_c_sdk REQUIRED)
foreach(tool mujoco_sim_ctl igris_ctl)
  add_executable(${tool} src/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE igris_c_sdk::igris_c_sdk)
  install(TARGETS ${tool} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
